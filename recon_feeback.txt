#!/bin/sh

CONFIG_FILE="config.cfg"
LOG_FILE="recon_check.log"

if [ $# -ne 1 ]; then
    echo "Usage: $0 <INTERFACE_NAME>"
    exit 1
fi

TARGET_IFACE="$1"

log() {
    echo "`date '+%Y-%m-%d %H:%M:%S'` - $*" >> "$LOG_FILE"
}

# Strip everything except digits (CR, LF, tabs, BOM, spaces, garbage)
digits_only() {
    sed 's/[^0-9]//g' "$1"
}

get_murex_value() {
    file="$1"
    val=`grep -i 'rowcount=' "$file" | awk -F= '{print $2}' | sed 's/[^0-9]//g'`
    echo "$val"
}

get_hub_value() {
    file="$1"
    val=`digits_only "$file"`
    echo "$val"
}

FOUND_SECTION=0

while IFS= read -r line; do

    case "$line" in
        \#*)
            header="${line#\#}"
            if [ "$header" = "$TARGET_IFACE" ]; then
                FOUND_SECTION=1
                continue
            fi
            if [ $FOUND_SECTION -eq 1 ]; then
                break
            fi
            ;;
    esac

    [ $FOUND_SECTION -eq 0 ] && continue
    [ -z "$line" ] && break

    iface=`echo "$line" | awk -F'|' '{print $1}'`
    murex_file=`echo "$line" | awk -F'|' '{print $2}'`
    hub_file=`echo "$line" | awk -F'|' '{print $3}'`
    archive_dir=`echo "$line" | awk -F'|' '{print $4}'`
    skiprows=`echo "$line" | awk -F'|' '{print $5}'`

    log "Processing interface: $iface"

    [ ! -f "$murex_file" ] && log "Missing murex file" && exit 1
    [ ! -f "$hub_file" ] && log "Missing hub feedback file" && exit 1

    murex_val=`get_murex_value "$murex_file"`
    hub_val=`get_hub_value "$hub_file"`

    echo "$murex_val" | grep -q '^[0-9][0-9]*$' || { log "Invalid murex_val"; exit 1; }
    echo "$hub_val" | grep -q '^[0-9][0-9]*$' || { log "Invalid hub_val"; exit 1; }
    echo "$skiprows" | grep -q '^[0-9][0-9]*$' || { log "Invalid skiprows"; exit 1; }

    # Perform subtraction (expr exit code is ignored)
    diff=`expr $murex_val - $hub_val - $skiprows 2>/dev/null`

    # REAL validation: diff must be numeric
    echo "$diff" | grep -q '^[0-9-][0-9-]*$' || {
        log "ERROR: expr produced invalid output"
        log "Values: murex=$murex_val hub=$hub_val skiprows=$skiprows"
        exit 1
    }

    log "murex=$murex_val hub=$hub_val skiprows=$skiprows diff=$diff"

    if [ "$diff" -ne 0 ]; then
        log "FAILED: diff=$diff"
        exit 1
    fi

    # SUCCESS → rename + archive
    ts=`date '+%Y%m%d_%H%M%S'`

    murex_base=`basename "$murex_file"`
    hub_base=`basename "$hub_file"`

    murex_new="${murex_base%.*}_$ts.${murex_base##*.}"
    hub_new="${hub_base%.*}_$ts.${hub_base##*.}"

    mv "$murex_file" "$archive_dir/$murex_new"
    mv "$hub_file" "$archive_dir/$hub_new"

    log "SUCCESS: archived $iface as $murex_new and $hub_new"

done < "$CONFIG_FILE"

echo "Processing completed for interface: $TARGET_IFACE"
