#!/bin/bash OUT="final_dashboard.html" cat < "$OUT" Enterprise Health Dashboard

Enterprise Environment Health Dashboard

EOF for file in env_*.json; do [ ! -f "$file" ] && continue ENV=$(grep '"environment"' "$file" | sed 's/.*: "\(.*\)".*/\1/') STATUS=$(grep '"global_status"' "$file" | sed 's/.*: "\(.*\)".*/\1/') ORCH=$(grep '"orchestrator"' "$file" | sed 's/.*: "\(.*\)".*/\1/') # FIXED: numeric-only extraction PIDS=$(grep '"pid_count"' "$file" | sed 's/[^0-9]//g') DISK_COUNT=$(grep '"disk_alerts"' "$file" | sed 's/[^0-9]//g') BADGE_CLASS=$(echo "$STATUS" | tr '[:upper:]' '[:lower:]') echo "
" >> "$OUT" echo "
$ENV $STATUS
" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" # Murex Services if [ "$PIDS" -gt 10 ]; then echo "" >> "$OUT" else echo "" >> "$OUT" fi # Disk summary if [ "$DISK_COUNT" -gt 0 ]; then echo "" >> "$OUT" else echo "" >> "$OUT" fi echo "
ComponentNodeStatusDetails
Murex Services$ORCHRUNNING$PIDS PIDs
Murex Services$ORCHSTOPPED$PIDS PIDs
DiskLocalALERT$DISK_COUNT partitions above threshold
DiskLocalSTABLENo issues
" >> "$OUT" # Collapsible Disk Details echo "
Disk Details ($DISK_COUNT alerts)" >> "$OUT" echo "" >> "$OUT" awk '/"disk_details"/,/]/' "$file" | grep '{' | while read -r line; do MOUNT=$(echo "$line" | sed 's/.*"mount": "\(.*\)".*/\1/') USAGE=$(echo "$line" | sed 's/.*"usage": "\(.*\)".*/\1/') echo "" >> "$OUT" done echo "
$MOUNT$USAGEHigh usage
" >> "$OUT" # Collapsible BL Servers echo "
Business Layer Servers" >> "$OUT" echo "" >> "$OUT" awk '/"bl_servers"/,/]/' "$file" | grep '{' | while read -r line; do NAME=$(echo "$line" | sed 's/.*"name": "\(.*\)".*/\1/') ST=$(echo "$line" | sed 's/.*"status": "\(.*\)".*/\1/') if [ "$ST" = "DOWN" ]; then echo "" >> "$OUT" else echo "" >> "$OUT" fi done echo "
$NAMEDOWNSSH timeout
$NAMEREACHABLEOK
" >> "$OUT" echo "
" >> "$OUT" done cat <> "$OUT" EOF echo "Dashboard generated: $OUT"
