#!/bin/bash OUT="final_dashboard.html" ############################################################ # HTML HEADER + MODERN COMPACT CSS ############################################################ cat < "$OUT" Enterprise Health Dashboard

Enterprise Environment Health Dashboard

EOF ############################################################ # LOOP THROUGH JSON FILES ############################################################ for file in env_*.json; do [ ! -f "$file" ] && continue ENV=$(grep '"environment"' "$file" | sed 's/.*: "\(.*\)".*/\1/') STATUS=$(grep '"global_status"' "$file" | sed 's/.*: "\(.*\)".*/\1/') OWNER=$(grep '"owner"' "$file" | sed 's/.*: "\(.*\)".*/\1/') VERSION=$(grep '"version"' "$file" | sed 's/.*: "\(.*\)".*/\1/') REFRESH=$(grep '"timestamp"' "$file" | sed 's/.*: "\(.*\)".*/\1/') PIDS=$(grep '"pid_count"' "$file" | sed 's/[^0-9]//g') DISK_COUNT=$(grep '"disk_alerts"' "$file" | sed 's/[^0-9]//g') case "$STATUS" in HEALTHY) BADGE_CLASS="ok" ;; WARNING) BADGE_CLASS="warn" ;; CRITICAL) BADGE_CLASS="crit" ;; *) BADGE_CLASS="warn" ;; esac echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" echo "" >> "$OUT" done ############################################################ # CLOSE HTML ############################################################ cat <> "$OUT"
Env St Owner Ver Refreshed PIDs Disk More
$ENV${OWNER:-N/A}${VERSION:-N/A}$REFRESH$PIDS$DISK_COUNT
View
" >> "$OUT" ############################################################ # DISK DETAILS ############################################################ echo "
Disk Details
" >> "$OUT" echo "" >> "$OUT" awk '/"disk_details"/,/]/' "$file" | grep '{' | while read -r line; do MOUNT=$(echo "$line" | sed 's/.*"mount": "\(.*\)".*/\1/') USAGE=$(echo "$line" | sed 's/.*"usage": "\(.*\)".*/\1/') echo "" >> "$OUT" done echo "
MountUsage
$MOUNT$USAGE
" >> "$OUT" ############################################################ # BL SERVERS ############################################################ echo "
BL Servers
" >> "$OUT" echo "" >> "$OUT" awk '/"bl_servers"/,/]/' "$file" | grep '{' | while read -r line; do NAME=$(echo "$line" | sed 's/.*"name": "\(.*\)".*/\1/') ST=$(echo "$line" | sed 's/.*"status": "\(.*\)".*/\1/') case "$ST" in REACHABLE) CLASS="ok" ;; DOWN) CLASS="crit" ;; *) CLASS="warn" ;; esac echo "" >> "$OUT" done echo "
ServerStatus
$NAME
" >> "$OUT" echo "
EOF echo "Dashboard generated: $OUT"
