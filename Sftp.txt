#!/bin/bash

###############################################################################
# Script Name : generic_sftp.sh
# Purpose     : Generic SFTP Transfer Utility
###############################################################################

SCRIPT_NAME=$(basename "$0" .sh)
TIMESTAMP=$(date '+%Y%m%d_%H%M%S')
LOG_FILE="${SCRIPT_NAME}_${TIMESTAMP}.log"

CONFIG_FILE="./sftp_config.cfg"

###############################################################################
# Logging Function
###############################################################################
log_msg()
{
    local MSG="$1"
    echo "$(date '+%Y-%m-%d %H:%M:%S') : ${MSG}" | tee -a "${LOG_FILE}"
}

###############################################################################
# Validation
###############################################################################
if [ $# -ne 1 ]; then
    echo "Usage: $0 <filename>"
    exit 1
fi

INPUT_FILE="$1"

if [ ! -f "${CONFIG_FILE}" ]; then
    echo "Config file ${CONFIG_FILE} not found."
    exit 1
fi

log_msg "============================================================"
log_msg "Script Started"
log_msg "Input Filename : ${INPUT_FILE}"
log_msg "Config File    : ${CONFIG_FILE}"
log_msg "Log File       : ${LOG_FILE}"
log_msg "============================================================"

###############################################################################
# Process Config Entries
###############################################################################
while IFS='|' read -r FILE_PATTERN SOURCE_DIR DEST_SERVER DEST_USER DEST_DIR
do

    # Skip blank lines
    [ -z "${FILE_PATTERN}" ] && continue

    # Skip comments
    [[ "${FILE_PATTERN}" =~ ^# ]] && continue

    log_msg "------------------------------------------------------------"
    log_msg "Processing Config Entry"
    log_msg "Pattern         : ${FILE_PATTERN}"
    log_msg "Source Location : ${SOURCE_DIR}"
    log_msg "Destination Host: ${DEST_SERVER}"
    log_msg "Destination User: ${DEST_USER}"
    log_msg "Destination Dir : ${DEST_DIR}"
    log_msg "------------------------------------------------------------"

    ###########################################################################
    # Check if filename matches configured pattern
    ###########################################################################
    if [[ "${INPUT_FILE}" == ${FILE_PATTERN} ]]; then

        FULL_FILE_PATH="${SOURCE_DIR}/${INPUT_FILE}"

        log_msg "Filename matches pattern ${FILE_PATTERN}"

        if [ ! -f "${FULL_FILE_PATH}" ]; then
            log_msg "ERROR: Source file not found: ${FULL_FILE_PATH}"
            continue
        fi

        log_msg "Source file found: ${FULL_FILE_PATH}"
        log_msg "Starting SFTP transfer..."

        #######################################################################
        # Execute SFTP
        #######################################################################
        sftp -b - "${DEST_USER}@${DEST_SERVER}" >> "${LOG_FILE}" 2>&1 <<EOF
cd ${DEST_DIR}
put ${FULL_FILE_PATH}
bye
EOF

        SFTP_RC=$?

        #######################################################################
        # Check Return Code
        #######################################################################
        if [ ${SFTP_RC} -eq 0 ]; then
            log_msg "SUCCESS: File transferred successfully."
            log_msg "File : ${FULL_FILE_PATH}"
            log_msg "Host : ${DEST_SERVER}"
        else
            log_msg "FAILURE: SFTP transfer failed."
            log_msg "Return Code : ${SFTP_RC}"
            log_msg "File        : ${FULL_FILE_PATH}"
            log_msg "Host        : ${DEST_SERVER}"
        fi

    else
        log_msg "Filename ${INPUT_FILE} does not match pattern ${FILE_PATTERN}"
    fi

done < "${CONFIG_FILE}"

log_msg "============================================================"
log_msg "Script Completed"
log_msg "============================================================"

exit 0
