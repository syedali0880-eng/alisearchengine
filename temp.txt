#!/bin/sh

CONFIG_FILE="extraction.cfg"
LOG_FILE="extraction_monitor.log"
TMP_TABLE="/tmp/rowcount_table_$$.txt"

MAIL_TO="ops-team@example.com"
MAIL_FROM="monitor@example.com"
MAIL_SUBJECT="Row Count Anomaly Report"

# ---------------------------------------------------------
# Logging
# ---------------------------------------------------------
log() {
    echo "`date '+%Y-%m-%d %H:%M:%S'` - $*" >> "$LOG_FILE"
}

# ---------------------------------------------------------
# Rowcount function (POSIX-safe)
# ---------------------------------------------------------
get_rowcount() {
    file="$1"

    # Extract extension safely
    ext=`echo "$file" | awk -F. '{print $NF}'`

    if [ "$ext" = "xls" ]; then
        grep -c '<Row ' "$file" 2>/dev/null
    else
        wc -l < "$file" 2>/dev/null
    fi
}

# ---------------------------------------------------------
# Initialize plain-text table
# ---------------------------------------------------------
echo "REPORT NAME                 | AVERAGE COUNT | TODAY'S COUNT | STATUS" > "$TMP_TABLE"
echo "-----------------------------------------------------------------------" >> "$TMP_TABLE"

# ---------------------------------------------------------
# Process each config line
# ---------------------------------------------------------
process_line() {
    pattern="$1"
    current_dir="$2"
    archive_dir="$3"

    log "Processing pattern=$pattern"

    # Find newest current file
    current_file=`ls -t "$current_dir"/$pattern 2>/dev/null | head -n 1`

    if [ -z "$current_file" ]; then
        log "No current file found for $pattern"
        return
    fi

    current_rows=`get_rowcount "$current_file"`

    if [ -z "$current_rows" ]; then
        log "Rowcount failed for $current_file"
        return
    fi

    log "Current rows: $current_rows"

    # Get last 7 archive files
    archive_files=`ls -t "$archive_dir"/$pattern 2>/dev/null | head -n 7`

    if [ -z "$archive_files" ]; then
        log "No archive files found for $pattern"

        printf "%-25s | %-13s | %-13s | %s\n" "$pattern" "$current_rows" "N/A" "NO BASELINE" >> "$TMP_TABLE"
        return
    fi

    sum=0
    count=0

    for f in $archive_files; do
        rc=`get_rowcount "$f"`
        if [ -n "$rc" ]; then
            sum=`expr $sum + $rc`
            count=`expr $count + 1`
            log "Archive $f rows=$rc"
        fi
    done

    if [ "$count" -eq 0 ]; then
        log "No valid archive rowcounts"

        printf "%-25s | %-13s | %-13s | %s\n" "$pattern" "$current_rows" "N/A" "NO BASELINE" >> "$TMP_TABLE"
        return
    fi

    avg=`expr $sum / $count`
    threshold=`expr $avg \* 80 / 100`

    log "Average=$avg Threshold=$threshold"

    if [ "$current_rows" -lt "$threshold" ]; then
        status="ALERT"
    else
        status="OK"
    fi

    printf "%-25s | %-13s | %-13s | %s\n" "$pattern" "$avg" "$current_rows" "$status" >> "$TMP_TABLE"
}

# ---------------------------------------------------------
# Main
# ---------------------------------------------------------
log "===== Script started ====="

if [ ! -f "$CONFIG_FILE" ]; then
    log "Config file missing: $CONFIG_FILE"
    exit 1
fi

# Read config file
while IFS='|' read pattern current_dir archive_dir; do
    case "$pattern" in
        ""|"#"*) continue ;;
    esac
    process_line "$pattern" "$current_dir" "$archive_dir"
done < "$CONFIG_FILE"

# ---------------------------------------------------------
# Send email (plain text always works)
# ---------------------------------------------------------
(
    echo "From: $MAIL_FROM"
    echo "To: $MAIL_TO"
    echo "Subject: $MAIL_SUBJECT"
    echo ""
    echo "Row Count Report"
    echo "=============================="
    echo ""
    cat "$TMP_TABLE"
) | /usr/sbin/sendmail -t

rm -f "$TMP_TABLE"

log "===== Script completed ====="
