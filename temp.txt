#!/bin/sh

CONFIG_FILE="extraction.cfg"
LOG_FILE="extraction_monitor.log"
TMP_TABLE="/tmp/rowcount_table_$$.txt"
TMP_SORT="/tmp/rowcount_sort_$$.txt"

MAIL_TO="ops-team@example.com"
MAIL_FROM="monitorreport@noextension.com"
MAIL_SUBJECT="Row Count Anomaly Report"

log() {
    echo "`date '+%Y-%m-%d %H:%M:%S'` - $*" >> "$LOG_FILE"
}

get_rowcount() {
    file="$1"
    ext=`echo "$file" | awk -F. '{print $NF}'`

    if [ "$ext" = "xls" ]; then
        grep -c '<Row ' "$file" 2>/dev/null
    else
        wc -l < "$file" 2>/dev/null
    fi
}

# Determine max width for dynamic column sizing
MAX_WIDTH=25

update_width() {
    len=`echo "$1" | wc -c`
    if [ "$len" -gt "$MAX_WIDTH" ]; then
        MAX_WIDTH="$len"
    fi
}

# First pass: scan config and update width
while IFS='|' read pattern current_dir archive_dir; do
    case "$pattern" in ""|"#"*) continue ;; esac
    update_width "$pattern"
done < "$CONFIG_FILE"

# Build dynamic border
BORDER="+"
i=1
while [ $i -le $MAX_WIDTH ]; do BORDER="${BORDER}-"; i=`expr $i + 1`; done
BORDER="${BORDER}+---------------+---------------+--------+"

# Header
echo "$BORDER" > "$TMP_SORT"
printf "| %-*s | %-13s | %-13s | %-6s |\n" "$MAX_WIDTH" "REPORT NAME" "AVERAGE COUNT" "TODAY'S COUNT" "STATUS" >> "$TMP_SORT"
echo "$BORDER" >> "$TMP_SORT"

process_line() {
    pattern="$1"
    current_dir="$2"
    archive_dir="$3"

    current_file=`ls -t "$current_dir"/$pattern 2>/dev/null | head -n 1`
    if [ -z "$current_file" ]; then return; fi

    current_rows=`get_rowcount "$current_file"`
    if [ -z "$current_rows" ]; then return; fi

    archive_files=`ls -t "$archive_dir"/$pattern 2>/dev/null | head -n 7`

    if [ -z "$archive_files" ]; then
        printf "1|%-*s|N/A|%s|NO BL\n" "$MAX_WIDTH" "$pattern" "$current_rows" >> "$TMP_SORT"
        return
    fi

    sum=0
    count=0
    for f in $archive_files; do
        rc=`get_rowcount "$f"`
        if [ -n "$rc" ]; then
            sum=`expr $sum + $rc`
            count=`expr $count + 1`
        fi
    done

    if [ "$count" -eq 0 ]; then
        printf "1|%-*s|N/A|%s|NO BL\n" "$MAX_WIDTH" "$pattern" "$current_rows" >> "$TMP_SORT"
        return
    fi

    avg=`expr $sum / $count`
    threshold=`expr $avg \* 80 / 100`

    if [ "$current_rows" -lt "$threshold" ]; then
        status="[ALERT]"
        sortkey=0
    else
        status="[OK]"
        sortkey=1
    fi

    printf "%s|%-*s|%s|%s|%s\n" "$sortkey" "$MAX_WIDTH" "$pattern" "$avg" "$current_rows" "$status" >> "$TMP_SORT"
}

# Second pass: process lines
while IFS='|' read pattern current_dir archive_dir; do
    case "$pattern" in ""|"#"*) continue ;; esac
    process_line "$pattern" "$current_dir" "$archive_dir"
done < "$CONFIG_FILE"

# Sort ALERT first
sort -n "$TMP_SORT" | awk -F"|" '
BEGIN { }
{
    if (NR == 1 || NR == 2) {
        print $0
    } else {
        printf "| %-s | %-13s | %-13s | %-6s |\n", $2, $3, $4, $5
    }
}
' > "$TMP_TABLE"

# Add final border
echo "$BORDER" >> "$TMP_TABLE"

# Send email
(
    echo "From: $MAIL_FROM"
    echo "To: $MAIL_TO"
    echo "Subject: $MAIL_SUBJECT"
    echo ""
    echo "Row Count Report"
    echo "=============================="
    echo ""
    cat "$TMP_TABLE"
) | /usr/sbin/sendmail -t

rm -f "$TMP_TABLE" "$TMP_SORT"
